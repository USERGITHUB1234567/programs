namespace soupnq{
    bool used[50];
    unordered_map<int,int>cnt;
    array<int,3>qu[maxn];
    long long ans[maxn],cur=0;
    const int block_size=317;
    void add(int i) {
        ++cnt[a[i]];

    }
    void rem(int i) {
        --cnt[a[i]];
    }
    void implement() {
        for(int i=1; i<=n; ++i) {
            auto[l,r]=query[i];
            qu[i][0]=l,qu[i][1]=r,qu[i][2]=i;
        }
        sort(qu+1,qu+1+q,[&](const array<int,3>& a,const array<int,3>& b){
            int bl_a=a[0]/block_size,bl_b=b[0]/block_size;
            return (bl_a<bl_b) || (bl_a==bl_b && a[1]<b[1]);
        });
        int ql=0,qr=-1;
        for(int t=1; t<=q; ++t) {
            auto[l,r,id]=qu[t];
            while(qr<r) add(++qr);
            while(ql<l) rem(ql++);
            while(qr>r) rem(qr--);
            while(ql>l) add(--ql);
            long long res=0;
            //cout << l << ' ' << r << '\n';
            //for(int i=1; i<=10; ++i) cout << cnt[i] << ' ';
            //cout << '\n';
            for(int i=1; i<50; ++i) used[i]=false;
            for(int f=1; f<50; ++f) {
                if(used[f]) continue;
                for(int i=l; i<=r; ++i) {
                    if(fib[f]<=a[i] || used[f]) continue;
                    int z=fib[f]-a[i];
                    if((a[i]==z && cnt[z]>=2) || (a[i]!=z && cnt[z])) {
                        ++res;used[f]=true;
                        //cout << fib[f] << ' ' << f << '\n';
                        //cout << i << ' ' << a[i] << ' ' << z << ' ' << cnt[z] << ' ' << fib[f] << ' ' << f << '\n';
                    }
                }
            }
            cout << '\n';
            ans[id]=res;
        }
        for(int i=1; i<=q; ++i) cout << ans[i] << '\n';
    }
}
